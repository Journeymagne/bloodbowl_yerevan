## Решения этого плана (контроллер подтверждает)

- `team_ledger.delta` и `balance_after` — `numeric(12,3)` в тысячах, а не `integer`: в коде деньги дробные (`src/domain/money.mjs`). Спека §3.2 правится в Task 5.

- Остаток пишется как `admin` только в админском маршруте `PATCH /api/admin/teams/:id`; проверки стадии там нет (админ исправляет ошибки).
- Итоги пишут в журнал и операции шага «Состав» (`hire`, `sale`, `staff`), иначе строки не складываются в разницу казны. Нулевые строки не пишутся.
- Строки журнала получают `created_at = clock_timestamp()`: строки одной транзакции сохраняют порядок для постраничной истории этапа 03.
- Павший (живые итоги и `015`) — игрок состава с травмой `dead`, которого нет в составе после итогов; снимок — игрок до матча (`before_roster` / `baseRoster`), причина `NULL` (в состояниях 2.x её нет).
- SQL миграций тестируется на настоящем Postgres: новые опциональные тесты (`GATA_TEST_DATABASE_URL`), без переменной они пропускаются.
- `absence` есть у каждого игрока (по умолчанию `null`) и обнуляется, когда игрок не пропускает матч; `roll` на прокачке необязателен и в нормализованном виде отсутствует, если не задан — закреплённые фикстуры не меняются.

## Порядок миграций 014 и 015

`readMigrations()` сортирует файлы по имени, `migrationStatus()` считает ожидающими **все** файлы с диска, которых нет в `schema_migrations`, и `applyMigrations()` применяет их в порядке имён. «Последней применённой» отметки нет. Поэтому:

- dev-база и репетиционные базы, получившие `015` в этом этапе, получат `014` при следующем `db:migrate` после 02b; сервер до этого не стартует (`assertMigrationsApplied`), как и с любой ожидающей миграцией;
- на проде в день X все семь файлов применяются одним запуском в порядке 009…015, то есть `014` раньше `015`;
- `deploy:migrate`, `assertMigrationsApplied` и `rehearsal:check` (`addedMigrations`) сравнивают имена, а не позиции.

Это безопасно при одном условии, которое план 02b обязан повторить: `014` не читает ничего, что пишет `015` (`team_fallen_players`), а `015` не читает `match_events`. `015` зависит только от таблиц 007 и 011. В этом этапе `015` создаётся последним, после `012` и `013`, так что не по порядку применяется только `014`.

- [ ] **Step 1: Помощник `test/helpers/scratch-db.mjs`**

```js
/**
 * A throwaway database on the local Postgres, for the tests that need the real
 * engine: migrations, and SQL a fake client would only echo back.
 *
 * Opt-in. GATA_TEST_DATABASE_URL names the local server — the DATABASE_URL in
 * .env will do, only its host, port and credentials are used. Without it these
 * tests skip, so `npm test` stays green on a machine with no database. The URL
 * must be local (scripts/rehearsal/guard.mjs); every call creates its own
 * gata_test_* database from template0 and drops it afterwards, pass or fail.
 */
import { randomBytes } from "node:crypto";
import pg from "pg";

import { migrationStatus, readMigrations } from "../../server/db/migrate.mjs";
import { assertLocalDatabaseUrl } from "../../scripts/rehearsal/guard.mjs";

const configured = process.env.GATA_TEST_DATABASE_URL;

/** `skip` for node:test: false when a database is configured, the reason otherwise. */
export const scratchSkip = configured ? false : "set GATA_TEST_DATABASE_URL to a local Postgres to run the database tests";

function urlFor(database) {
  const url = assertLocalDatabaseUrl(configured);
  url.pathname = `/${database}`;
  return url.toString();
}

/** One migration file as the runner applies it: its SQL and its log row, in one transaction. */
export async function runMigration(client, name) {
  const migration = (await readMigrations()).find((item) => item.name === name);
  if (!migration) throw new Error(`no migration ${name}`);
  await client.query("BEGIN");
  try {
    await client.query(migration.sql);
    await client.query("INSERT INTO schema_migrations (name, checksum) VALUES ($1, $2) ON CONFLICT (name) DO NOTHING",
      [migration.name, migration.checksum]);
    await client.query("COMMIT");
  } catch (error) {
    await client.query("ROLLBACK").catch(() => {});
    throw error;
  }
}

/** Every migration up to and including `last` (all of them without it), in the runner's order. */
export async function migrateThrough(client, last = null) {
  await migrationStatus(client); // creates schema_migrations exactly as the runner does
  for (const { name } of await readMigrations()) {
    if (last && name.localeCompare(last, "en") > 0) break;
    await runMigration(client, name);
  }
}

export async function withScratchDatabase(run, { through = null } = {}) {
  const name = `gata_test_${randomBytes(6).toString("hex")}`;
  const server = new pg.Client({ connectionString: urlFor("postgres") });
  await server.connect();
  try {
    // template0: nobody is ever connected to it, so parallel test files never collide on the template.
    await server.query(`CREATE DATABASE ${name} TEMPLATE template0`);
    const client = new pg.Client({ connectionString: urlFor(name) });
    await client.connect();
    try {
      await migrateThrough(client, through);
      await run({ client, url: urlFor(name) });
    } finally {
      await client.end();
    }
  } finally {
    await server.query(`DROP DATABASE IF EXISTS ${name} WITH (FORCE)`)
      .catch((error) => console.error(`could not drop ${name}: ${error.message}`));
    await server.end();
  }
}
```

```js
export function classifyTreasuryChange(team, before, after, { admin = false } = {}) {
  const entries = team ? [...playerEntries(team, before, after), ...staffEntries(before, after)] : [];
  const change = sumMoney(countToNumber(after.treasury), -countToNumber(before.treasury));
  const residual = sumMoney(change, ...entries.map((entry) => -entry.delta));
  if (residual) entries.push({ kind: admin ? "admin" : "manual", delta: residual });
  return entries;
}

function playerEntries(team, before, after) {
  const was = normalizeDraftPlayers(team, before), now = normalizeDraftPlayers(team, after);
  const had = new Set(was.map((player) => player.id)), kept = new Set(now.map((player) => player.id));
  const hires = now.filter((player) => !had.has(player.id) && playerPurchaseRefund(player) !== null)
    .map((player) => ({ kind: "hire", delta: -playerPurchaseRefund(player) }));
  const removals = was.filter((player) => !kept.has(player.id)).map((player) => {
    const refund = playerPurchaseRefund(player);
    return refund !== null ? { kind: "cancel_purchase", delta: refund } : { kind: "sale", delta: playerSaleValue(team, player) };
  });
  return [...hires, ...removals].filter((entry) => entry.delta);
}
```

Шапка: таблицы бросков лиги (D8 из Player Advancement, Skill Table); случайный навык — основной (Random в таблице прокачки — «Randomly Select a Primary Skill»), грань `1…N` по колонке категории в порядке Skill Table; уже имеющиеся навыки (`skillNamesForPlayer`: базовые, купленные, Favoured, Pro капитана) перебрасываются, все попытки записываются; генератор передаётся снаружи.
```js
export function randomSkillRoll({ row, player, category, skillGroups = [], rng, maxAttempts = 100 }) {
  if (!categoriesForAccess(row.primary ?? []).includes(category)) return { ok: false, reason: "DICE_CATEGORY" };
  const candidates = [...(skillGroups.find((group) => group.category === category)?.skills ?? [])];
  const held = new Set(skillNamesForPlayer(row, player));
  if (!candidates.some((skill) => !held.has(skill))) return { ok: false, reason: "DICE_NO_SKILL" };
  const attempts = [];
  while (attempts.length < maxAttempts) {
    const roll = face(rng, candidates.length), skill = candidates[roll - 1], duplicate = held.has(skill);
    attempts.push({ roll, skill, duplicate });
    if (!duplicate) return { ok: true, params: { category, candidates }, result: { attempts, skill } };
  }
  throw new Error(`no free skill after ${maxAttempts} attempts`);
}
```

`face(rng, sides)` — локальная: значение `rng(sides)`, не целое в `1…sides` → `throw new RangeError(\`a D${sides} cannot show ${value}\`)`.

```sql
-- 015_backfill_fallen_players.sql — players who died in a match finished
-- through the post-match checklist before 3.0.
--
-- 2.x dropped a dead player from the roster and kept no other trace of them,
-- except in the post-match itself: the side's injuries in match_post_games.state
-- (keyed by player id), and the roster before and after in
-- match_post_applications. A player counts as fallen when their injury there is
-- `dead`, they were on the roster before the match, and they are not on it
-- after. The last condition is rookie protection: a team with fewer than three
-- games turns any injury into Badly Hurt (effectiveInjury in
-- src/domain/post-match/rules.mjs) and keeps the player. Journeymen and star
-- players were never on the roster, so they are never carried over.
--
-- The snapshot is the player as before_roster holds them; died_at is when the
-- post-match was applied. 2.x states record no cause, so cause is NULL, and
-- recorded_by is NULL because the match recorded it, not a person. A row the
-- 3.0 post-match already wrote for the same team, match and player is not
-- repeated, so a second run adds nothing.
--
-- scripts/rehearsal/snapshot.mjs (fallenCheck) applies the same rule to check
-- this migration: change both together. 014 (match_events, stage 02b) is
-- numbered before this file but written after it; the runner applies whatever
-- is missing in name order, and neither file reads what the other writes.
INSERT INTO team_fallen_players (saved_team_id, player, cause, pairing_id, recorded_by, died_at)
SELECT a.saved_team_id, p.player, NULL, a.pairing_id, NULL, a.created_at
  FROM match_post_applications a
  JOIN match_post_games g ON g.pairing_id = a.pairing_id
 CROSS JOIN LATERAL jsonb_array_elements(CASE WHEN jsonb_typeof(a.before_roster->'players') = 'array'
                                              THEN a.before_roster->'players' ELSE '[]'::jsonb END) AS p(player)
 WHERE g.state->a.side->'injuries'->(p.player->>'id')->>'code' = 'dead'
   AND NOT EXISTS (
         SELECT 1 FROM jsonb_array_elements(CASE WHEN jsonb_typeof(a.after_roster->'players') = 'array'
                                                 THEN a.after_roster->'players' ELSE '[]'::jsonb END) AS q(player)
          WHERE q.player->>'id' = p.player->>'id')
   AND NOT EXISTS (
         SELECT 1 FROM team_fallen_players f
          WHERE f.saved_team_id = a.saved_team_id AND f.pairing_id = a.pairing_id
            AND f.player->>'id' = p.player->>'id');
```
